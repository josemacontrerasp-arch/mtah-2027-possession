# or with plain pip (inside a Python 3.12 virtual environment):
#   python -m pip install -r requirements-solution.txt -c requirements-solution.lock.txt

# runtime (a large download), the CPU-only wheel can be installed first, then the line above:
#   uv pip install --python .venv --no-deps torch==2.11.0 --index-url https://download.pytorch.org/whl/cpu
